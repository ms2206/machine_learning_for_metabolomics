# rename Storage.Week levels to valid R variable names
training_features$Storage.Week <- factor(
  training_features$Storage.Week,
  levels = levels(training_features$Storage.Week),
  labels = paste0("Week_", levels(training_features$Storage.Week))
)
# use stratified k-fold cross-validation (ensures all weeks in each fold)
train_control <- trainControl(
  method = "cv",
  number = 5,
  classProbs = TRUE,
  summaryFunction = multiClassSummary,
  savePredictions = "final",
  index = createFolds(training_features$Storage.Week, k = 5, list = TRUE, returnTrain = TRUE)
)

set.seed(123) # for reproducibility
# Define parameter grid
tune_grid <- expand.grid(mtry = c(5, 10, 15, 20, 25))

# Train with grid search
rf_model <- train(
  Storage.Week ~ .,
  data = training_features,
  method = "rf",
  trControl = train_control,
  tuneGrid = tune_grid,
  metric = "Accuracy",
  ntree = 1000,
  importance = TRUE,
